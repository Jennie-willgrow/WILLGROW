-- ════════════════════════════════════════════════════════════ -- WILLGROW — 시간표를 두 벌 둘 수 있게 -- -- 왜 필요한가요? -- 과정이 바뀌는 반(Phonics → Reading 등)은 요일·교시 배치와 -- 교재가 통째로 달라집니다. -- 그때마다 시간표를 갈아엎으면 지금 진행 중인 계획이 흔들려요. -- -- 그래서 시간표에 "언제부터" 를 붙입니다. -- -- 3C 시간표 -- ├ 지금 쓰는 것 2026-03-02 부터 -- └ 다음 것 2027-01-05 부터 ← 미리 짜 둡니다 -- -- 레슨플랜은 날짜마다 그날 유효한 시간표를 보고 만듭니다. -- 전환일을 지나면 알아서 새 배치·새 교재로 이어집니다. -- -- 사용법 -- Supabase > SQL Editor 에 통째로 붙여넣고 Run -- -- 안전한가요? -- 네. 칸 하나를 더하고, 지금 있는 시간표는 전부 -- "처음부터 쓰는 것" 으로 표시할 뿐입니다. -- 여러 번 실행해도 괜찮아요. -- ════════════════════════════════════════════════════════════ -- 이 시간표를 언제부터 쓰는지. 비어 있으면 "처음부터(지금 것)". ALTER TABLE class_schedules ADD COLUMN IF NOT EXISTS effective_from DATE; -- ── 한 반이 같은 날짜에 같은 칸을 두 번 갖지 않게 ── -- 예전 자물쇠는 (반·요일·교시) 였습니다. -- 이제 시간표가 여러 벌이므로 "언제부터" 까지 묶어 잠급니다. DROP INDEX IF EXISTS uq_class_schedule_slot; CREATE UNIQUE INDEX uq_class_schedule_slot ON class_schedules(class_id, COALESCE(effective_from, DATE '1900-01-01'), weekday, period); CREATE INDEX IF NOT EXISTS idx_class_schedules_from ON class_schedules(class_id, effective_from); -- ════════════════════════════════════════════════════════════ -- 확인 -- ════════════════════════════════════════════════════════════ SELECT c.name AS "반", COALESCE(s.effective_from::text, '처음부터 (지금)') AS "언제부터", COUNT(*) AS "칸 수", string_agg(DISTINCT CASE s.weekday WHEN 1 THEN '월' WHEN 2 THEN '화' WHEN 3 THEN '수' WHEN 4 THEN '목' WHEN 5 THEN '금' ELSE '주말' END, '·' ORDER BY CASE s.weekday WHEN 1 THEN '월' WHEN 2 THEN '화' WHEN 3 THEN '수' WHEN 4 THEN '목' WHEN 5 THEN '금' ELSE '주말' END) AS "요일" FROM class_schedules s JOIN classes c ON c.id = s.class_id GROUP BY c.name, s.effective_from ORDER BY c.name, s.effective_from NULLS FIRST; -- ↑ 지금은 모든 반이 "처음부터 (지금)" 한 줄씩만 나옵니다. -- 다음 시간표는 앱의 [Class Hub] > 반 카드 > 시간표 에서 -- [＋ 다음 시간표] 로 만드시면 됩니다. -- ── 되돌리기 ── -- DROP INDEX IF EXISTS uq_class_schedule_slot; -- CREATE UNIQUE INDEX uq_class_schedule_slot -- ON class_schedules(class_id, weekday, period); -- ALTER TABLE class_schedules DROP COLUMN effective_from;
